cmake_minimum_required(VERSION 3.21)
project(GDVsZombies VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)

option(GVZ_TESTS_ONLY "Build the pure C++ simulation tests without Geode" OFF)
if(GVZ_TESTS_ONLY)
    enable_testing()
    add_executable(gvz_tests tests/game_tests.cpp src/Game.cpp)
    target_include_directories(gvz_tests PRIVATE src)
    add_test(NAME game_rules COMMAND gvz_tests)
    add_executable(gvz_balance tests/balance_smoke.cpp src/Game.cpp)
    target_include_directories(gvz_balance PRIVATE src)
    add_test(NAME complete_campaign COMMAND gvz_balance)
    return()
endif()

if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "Set GEODE_SDK to the Geode v5.10.1 SDK directory. See README_RU.txt.")
endif()
add_library(${PROJECT_NAME} SHARED src/main.cpp src/GardenLayer.cpp src/Game.cpp)
add_subdirectory("$ENV{GEODE_SDK}" "${CMAKE_CURRENT_BINARY_DIR}/geode")
setup_geode_mod(${PROJECT_NAME})
